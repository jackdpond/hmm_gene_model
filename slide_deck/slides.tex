\section{Introduction}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{The Yeast Metabolic Cycle}
  \begin{itemize}
    \item \textit{Saccharomyces cerevisiae} (budding yeast) exhibits a
          spontaneous, synchronized \textbf{metabolic oscillation} when grown in continuous culture.
    \item The cycle is visible in O$_2$ consumption and spans distinct
          respiratory and reductive phases.
    \item Thousands of genes are expressed periodically, in groups called \textbf{temporal modules}
  \end{itemize}

  \bigskip
  \begin{block}{Two Central Questions}
    \begin{enumerate}
      \item What is the period of the metabolic cycle, and what are its phases?
      \item Which genes drive each phase, and what biological processes do they represent?
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Dataset (maybe replace this slide with diagram)}
  \begin{itemize}
    \item \textbf{Source:} Tu et al.\ (2005), \textit{Science} — publicly
          available via GEO (GSM77298).
    \item \textbf{Assay:} Yeast whole-transcriptome microarray.
    \item \textbf{Features:} 9,725 RNA transcripts (6,400 annotated genes
          + controls).
    \item \textbf{Time points:} 36 samples at $\approx$25-minute intervals
          ($\approx$15 hours total).
    \item Each time point is an independent microarray chip measuring
          relative transcript abundance across the \emph{same} oscillating
          population.
  \end{itemize}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────
\section{Data Processing}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Preprocessing Pipeline}
  Raw intensities require three normalization steps before analysis.

  \medskip
  \begin{enumerate}
    \item \textbf{Per-chip median normalization.}
          Each chip is divided by its median intensity, removing global
          staining and loading differences between arrays.
          \begin{itemize}
            \item Assumption: true median expression is constant across time.
            \item Result: \emph{relative} (not absolute) expression.
          \end{itemize}

    \item \textbf{arcsinh transform. (maybe a graph would be good here)}
          Compresses the dynamic range (similar to $\log$, but handles zeros).
          \[
            \tilde{x} = \operatorname{arcsinh}(x) = \ln\!\bigl(x + \sqrt{x^2+1}\bigr)
          \]

    \item \textbf{Per-gene $z$-score} (for clustering only).
          Each gene's time series is standardized to zero mean and unit
          variance so genes with very different baseline expressions are
          comparable.
  \end{enumerate}
\end{frame}

\section{Phase I: Period Discovery}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Approach: Clustering Time Points}
  \begin{block}{Key Insight}
    If the metabolic cycle has period $T$, time points separated by $T$
    should look alike — they should fall in the \emph{same cluster}.
  \end{block}

  \medskip
  \textbf{Strategy:} Cluster the 36 \emph{time points} (not genes) using a
  Gaussian Mixture Model (GMM), then read off the period from the repeating
  cluster sequence.

  \medskip
  \textbf{Dimensionality reduction} (necessary — $n=36 \ll d=9725$):
  \begin{itemize}
    \item Approach 1: Keep top-$n$ highest-variance genes, sweep $n \in
          [200, 1100]$.
    \item Approach 2: Project all genes to 5 PCA components; select $k$ by
          held-out log-likelihood (first 24 train, last 12 test).
    \item Both approaches consistently recover $k=4$ clusters.
  \end{itemize}
\end{frame}

\begin{frame}{Result: Period = 12 Time Steps, 300 seconds}
  \begin{figure}
    \centering
    \includegraphics[width=0.48\linewidth]{../paper/figures/rough-cell-expression-cycles.png}
    \hfill
    \includegraphics[width=0.48\linewidth]{../paper/figures/rough-gmm-classes.png}
    \caption{Left: 36 time points in PCA space, colored by GMM cluster.
             Right: Cluster assignment vs.\ time step, showing three
             complete repetitions of the 4-phase cycle.}
  \end{figure}
\end{frame}

\begin{frame}{Period Discovery — Summary}
  \begin{itemize}
    \item Every dimensionality-reduction and GMM variant converged to a
          \textbf{12-timestep period}.
    \[
      T = 12 \times 25\,\text{min} = 300\,\text{min}
    \]
    \item Cluster assignments are near-deterministic (high posterior
          probabilities), indicating well-separated phases. Also curse of dimensionality.
    \item The 36-point experiment spans exactly \textbf{3 complete cycles}.
    \item Independent confirmation of the period reported by Tu et al.
  \end{itemize}
\end{frame}

\section{Phase II: Gene Analysis}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Identifying Periodic Genes}
  With period $T=12$ established, which genes actually oscillate?

  \medskip
  \textbf{Permutation test on lag-12 autocorrelation:}
  \begin{enumerate}
    \item Compute $\text{ACF}(x_g, \tau=12)$ for each gene $g$.
    \item Shuffle gene $g$'s time series 100 times; compute
          $\text{ACF}$ for each shuffle.
    \item Accept gene as periodic if its true ACF is at or above the
          \textbf{95th percentile} of the shuffled distribution.
  \end{enumerate}

  \medskip
  \textbf{Result:} 5,243 of 9,725 genes ($\approx$54\%) are identified as
  periodic at lag 12.
\end{frame}

\begin{frame}{Periodic vs.\ Non-Periodic Genes in PCA}

\end{frame}

\begin{frame}{Classical Time Series Decomposition}
  For each periodic gene, decompose its standardized time series:
  \[
    x_g(t) = \underbrace{m_g(t)}_{\text{trend}} +
             \underbrace{s_g(t)}_{\text{seasonal}} +
             \underbrace{\varepsilon_g(t)}_{\text{residual}}
  \]
  Trend estimated via centered moving average (24 interior time points).

  \medskip
  \textbf{Clustering the seasonal components:}
  \begin{itemize}
    \item Fit GMMs over $k = 2, \ldots, 12$; select $k$ by AIC/BIC.
    \item \textbf{$k=8$ minimizes BIC} and gives a local AIC minimum.
    \item 7 clusters show distinct temporal expression profiles;
          1 cluster is a catch-all for noisy genes.
  \end{itemize}
\end{frame}

\begin{frame}{Temporal Expression Modules}
  \begin{figure}
    \centering
    \includegraphics[width=\linewidth]{../paper/figures/rough-seasonal-clusters.png}
    \caption{Left: UMAP of genes by seasonal component, colored by GMM
             cluster. Right: Mean seasonal profile for each of the 7
             coherent clusters, showing peaks and troughs at distinct
             phases of the 12-step cycle.}
  \end{figure}
\end{frame}

\begin{frame}{ARMA Forecasting on Individual Genes}
  \begin{itemize}
    \item ARMA$(p,q)$ models fit to each periodic gene's time series.
    \item \textbf{Train:} first 24 time steps (2 full cycles).
          \textbf{Test:} last 12 time steps (1 held-out cycle).
    \item Order $(p,q)$ selected per gene by AIC.
    \item Models capture the periodic structure and \textbf{predict the
          final cycle} with reasonable accuracy for well-expressed,
          high-amplitude genes.
    \item Performance degrades for low-amplitude or noisy genes —
          motivating more expressive sequence models.
  \end{itemize}
\end{frame}

\begin{frame}{Continuous-Density HMM Prediction}
  \begin{itemize}
    \item A \textbf{Gaussian mixture HMM (CDHMM)} treats the hidden metabolic
          phase as a latent Markov chain and models observations with a GMM
          emission distribution.
    \item Hidden states correspond to the 4 metabolic phases recovered earlier.
    \item Trained on the first 24 time steps; evaluated on the final 12.
    \item Captures \textbf{phase transitions} explicitly — emission
          distributions encode which genes are up/down in each phase.
    \item Prediction: given observed gene expression, infer the current
          phase and project forward.
  \end{itemize}

\section{Semantic Annotation via LDA}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{What Are the Gene Clusters \emph{Doing}?}
  Clustering recovers \emph{when} groups of genes are expressed.
  To understand \emph{why}, we need biological annotation.

  \medskip
  \textbf{Approach:} Latent Dirichlet Allocation (LDA) on gene-level
  metadata (GO terms, functional annotations).
  \begin{itemize}
    \item Each gene is a ``document''; its annotations are ``words''.
    \item LDA discovers $K=10$ latent \textbf{functional topics}.
    \item Topic weights per gene are then correlated with the
          time-series clusters to ask: \emph{which biological processes
          dominate each temporal module?}
  \end{itemize}
\end{frame}

\section{Conclusions}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Summary}
  \begin{enumerate}
    \item \textbf{Period:} GMM clustering of time points robustly recovers
          $T = 12$ steps $= 300$ minutes across all dimensionality-reduction
          strategies.
    \item \textbf{Periodic genes:} 5,243/9,725 genes ($\approx$54\%)
          oscillate with this period (permutation test, $\alpha=0.05$).
    \item \textbf{Temporal modules:} Seasonal decomposition + GMM
          identifies 7 coherent clusters of co-expressed genes, each with
          a distinct phase profile.
    \item \textbf{Forecasting:} ARMA and CDHMM models capture the
          periodic dynamics and predict held-out expression.
    \item \textbf{Function:} LDA links each temporal module to specific
          biological processes — oxidative metabolism, translation,
          replication, degradation — giving a functional portrait of the
          yeast metabolic cycle.
  \end{enumerate}
\end{frame}

\begin{frame}{Backup: GMM Model Selection}
  \begin{itemize}
    \item Held-out log-likelihood evaluated over 100 random restarts for
          $k \in \{2,\ldots,7\}$.
    \item $k=4$ consistently maximized held-out likelihood.
    \item Tied covariance used (full covariance is ill-conditioned for
          $n=24$, $d=5$).
  \end{itemize}
\end{frame}

\begin{frame}{Backup: Permutation Test Details}
  For each gene $g$ with time series $\mathbf{x}_g$:
  \[
    r_g = \frac{1}{n - \tau}
          \sum_{t=1}^{n-\tau}
          \frac{(x_g(t) - \bar{x}_g)(x_g(t+\tau) - \bar{x}_g)}{s_g^2},
    \quad \tau = 12
  \]
  Null distribution: 100 random permutations of $\mathbf{x}_g$.\\
  Reject null if $r_g \geq$ 95th percentile of permuted $r_g^{(b)}$.
\end{frame}

\begin{frame}{Backup: LDA Generative Model}
  For each gene $g$:
  \begin{enumerate}
    \item Draw topic proportions $\bm{\theta}_g \sim \text{Dir}(\alpha)$.
    \item For each annotation term $w$:
      \begin{enumerate}
        \item Draw topic $z \sim \text{Categorical}(\bm{\theta}_g)$.
        \item Draw term $w \sim \text{Categorical}(\bm{\phi}_z)$.
      \end{enumerate}
  \end{enumerate}
  Inference via variational EM; $K=10$ topics selected by held-out
  perplexity. Topic names assigned by prompting Claude with top-20 terms
  per topic.
\end{frame}
