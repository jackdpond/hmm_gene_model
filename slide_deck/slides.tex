\begin{frame}{Model}
  Let $X_1, \dots, X_n$ be observations. We assume
  \[
    X_i \mid \theta \;\overset{\text{iid}}{\sim}\; p(x \mid \theta),
  \]
  and place a prior $\pi(\theta)$ on the parameter of interest.

  \medskip
  The posterior is then
  \[
    \pi(\theta \mid x_1, \dots, x_n)
      \;\propto\; \pi(\theta) \prod_{i=1}^{n} p(x_i \mid \theta).
  \]
\end{frame}
